export function inventoryAgentParams(modelId: string) {
  return { name: "Inventory Reader", model: modelId,
    mcp_servers: [{ type: "url" as const, name: "inventory", url: "https://inventory.example.com/mcp" }],
    tools: [{ type: "mcp_toolset" as const, mcp_server_name: "inventory",
      default_config: { enabled: false, permission_policy: { type: "always_ask" as const } },
      configs: [{ name: "get_stock", enabled: true }] }], skills: [], multiagent: null };
}
export async function createInventoryAgent(client: Anthropic, modelId: string) {
  // Placeholder endpoint; replace with your reviewed reachable HTTPS MCP service.

export function inventorySessionParams(agentId: string, version: number, environmentId: string, authorizedVaultId: string) {
  if (!authorizedVaultId.trim()) throw new Error("Authorized vault ID required");
  return { agent: { type: "agent" as const, id: agentId, version },
    environment_id: environmentId, vault_ids: [authorizedVaultId] };
}
export async function createInventorySession(client: Anthropic, agentId: string, version: number,
  environmentId: string, authorizedVaultId: string) {
  return client.beta.sessions.create(inventorySessionParams(agentId, version, environmentId, authorizedVaultId),
    { maxRetries: 0, timeout: 10000 });
}
export function diagnoseMcp(event: Event) {
  if (event.type !== "session.error") return null;
  const e = event.error;
  if (e.type !== "mcp_connection_failed_error" && e.type !== "mcp_authentication_failed_error") return null;
  return { eventId: event.id, kind: e.type, server: e.mcp_server_name, retry: e.retry_status.type };
}
export function makeNativeEvidence() {
  const calls = new Map<string, { name: string; server: string }>();
  return { ingest(event: Event) {
    if (event.type === "agent.mcp_tool_use") calls.set(event.id, { name: event.name, server: event.mcp_server_name });
    if (event.type !== "agent.mcp_tool_result") return null;
    const call = calls.get(event.mcp_tool_use_id);
    return { callId: event.mcp_tool_use_id, resultId: event.id, matched: !!call, call,
      outcome: event.is_error === true ? "error" : event.is_error === false ? "non-error-result" : "unspecified" };
  } };
}
const StockInput = z.strictObject({ sku: z.string().regex(/^SKU-[0-9]{4}$/) });
type LocalClient = { callTool: (request: { name: string; arguments: { sku: string } }, options: { signal: AbortSignal }) => Promise<unknown> };
export async function bridgeStock(name: string, input: unknown, context: { authorized: boolean; signal: AbortSignal }, mcp: LocalClient) {
  if (!context.authorized || name !== "get_stock") throw new Error("Bridge call not authorized or enabled");
  const args = StockInput.parse(input);
  context.signal.throwIfAborted();
  return mcp.callTool({ name: "get_stock", arguments: args }, { signal: context.signal });
}
